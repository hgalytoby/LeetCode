```py
class Logger:

    def __init__(self):
        self.limit = {}

    def shouldPrintMessage(self, timestamp: int, message: str) -> bool:
        if self.limit.get(message, 0) > timestamp:
            return False
        self.limit[message] = timestamp + 10
        return True
```

### Go
```go
type Logger struct {
	limit map[string]int
}

func Constructor() Logger {
	return Logger{
		limit: make(map[string]int),
	}
}

func (this *Logger) ShouldPrintMessage(timestamp int, message string) bool {
	if nextTime, ok := this.limit[message]; ok && nextTime > timestamp {
		return false
	}
	this.limit[message] = timestamp + 10
	return true
}


/**
 * Your Logger object will be instantiated and called as such:
 * obj := Constructor();
 * param_1 := obj.ShouldPrintMessage(timestamp,message);
 */
```

### C#
```c#
public class Logger {

    private Dictionary<string, int> limit;

    public Logger() {
        limit = new Dictionary<string, int>();
    }

    public bool ShouldPrintMessage(int timestamp, string message) {
        if (limit.TryGetValue(message, out int nextTime) && nextTime > timestamp)
        {
            return false;
        }
        limit[message] = timestamp + 10;
        return true;
    }
}

/**
 * Your Logger object will be instantiated and called as such:
 * Logger obj = new Logger();
 * bool param_1 = obj.ShouldPrintMessage(timestamp,message);
 */
```
