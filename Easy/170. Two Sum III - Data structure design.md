```py
class TwoSum:

    def __init__(self):
        self.counter = defaultdict(int)

    def add(self, number: int) -> None:
        self.counter[number] += 1
        
    def find(self, value: int) -> bool:
        for num, count in self.counter.items():
            diff = value - num
            if diff == num and count < 2:
                continue
            if diff in self.counter:
                return True
        return False
```

```go
type TwoSum struct {
    counter map[int]int
}


func Constructor() TwoSum {
    return TwoSum{counter: make(map[int]int)}
}


func (this *TwoSum) Add(number int)  {
    this.counter[number]++
}


func (this *TwoSum) Find(value int) bool {
    for num, count := range this.counter {
		diff := value - num
		if diff == num && count < 2 {
			continue
		}
		if _, exists := this.counter[diff]; exists {
			return true
		}
	}
	return false
}
```

### C#
```c#
public class TwoSum {
    private Dictionary<int, int> _counter;

    public TwoSum() {
        _counter = new Dictionary<int, int>();
    }
    
    public void Add(int number) {
        if (_counter.ContainsKey(number)) {
            _counter[number]++;
        } else {
            _counter[number] = 1;
        }
    }
    
    public bool Find(int value) {
        foreach (var pair in _counter) {
            int num = pair.Key;
            int count = pair.Value;
            int diff = value - num;
            if (diff == num && count < 2) {
                continue;
            }
            if (_counter.ContainsKey(diff)) {
                return true;
            }
        }
        return false;
    }
}

 * Your TwoSum object will be instantiated and called as such:
 * TwoSum obj = new TwoSum();
 * obj.Add(number);
 * bool param_2 = obj.Find(value);
 */
```
