### Python
```py
class Solution:
    def similarRGB(self, color: str) -> str:
        result = ""
        for i in range(1, 6, 2):
            c = int(color[i] + color[i + 1], 16)
            j = round(c / 17)
            result += hex(j * 17)[2:].zfill(2)
        return f"#{result}"
```

### C#
```c#
public class Solution {
    public string SimilarRGB(string color) {
        string result = "";
        for (int i = 1; i < 6; i += 2) {
            string subStr = color.Substring(i, 2);
            int c = Convert.ToInt32(subStr, 16);
            int j = (int)Math.Round((double)c / 17.0);
            result += (j * 17).ToString("x2");
        }
        return $"#{result}";
    }
}
```
