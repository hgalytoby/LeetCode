### Python
```py
class Solution:
    def highFive(self, items: List[List[int]]) -> List[List[int]]:
        group = defaultdict(list)
        for id_, score in items:
            group[id_].append(score)
        for v in group.values():
            v.sort()
        return sorted([[k, sum(v[-5:]) // 5] for k, v in group.items()])
```

### Go
```
func highFive(items [][]int) [][]int {
    group := make(map[int][]int)
	result := [][]int{}
	for _, item := range items {
		id := item[0]
		score := item[1]
		group[id] = append(group[id], score)
	}
	for _, v := range group {
		sort.Ints(v)
	}
	for k, v := range group {
		sum := 0
		start := len(v) - 5
		for i := start; i < len(v); i++ {
			sum += v[i]
		}
		avg := sum / 5
		result = append(result, []int{k, avg})
	}
	sort.Slice(result, func(i, j int) bool {
		return result[i][0] < result[j][0]
	})
	return result
}
```

### C#
```c#
public class Solution {
    public int[][] HighFive(int[][] items) {
        Dictionary<int, List<int>> group = new ();
        List<int[]> resList = new ();
        foreach (int[] item in items) {
            int id = item[0];
            int score = item[1];
            if (!group.ContainsKey(id)) {
                group[id] = new List<int>();
            }
            group[id].Add(score);
        }

        foreach (KeyValuePair<int, List<int>> kvp in group) {
            List<int> v = kvp.Value;
            int sum = 0;
            v.Sort();
            for (int i = v.Count - 5; i < v.Count; i++) {
                sum += v[i];
            }
            resList.Add(new int[] { kvp.Key, sum / 5 });
        }
        return resList.OrderBy(x => x[0]).ToArray();
    }
}
```
