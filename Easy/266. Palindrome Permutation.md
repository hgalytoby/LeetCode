### Python

class Solution:
    def canPermutePalindrome(self, s: str) -> bool:
        counter = Counter(s)
        flag = False
        for v in counter.values():
            if v % 2 == 1:
                if flag:
                    return False
                flag = True
        return True

public class Solution {
    public bool CanPermutePalindrome(string s) {
        Dictionary<char, int> counter = new ();
        bool flag = false;
        foreach (char c in s)
        {
            if (counter.ContainsKey(c))
            {
                counter[c]++;
            }
            else
            {
                counter[c] = 1;
            }
        }
        foreach (int v in counter.Values)
        {
            if (v % 2 == 1)
            {
                if (flag)
                {
                    return false;
                }
                flag = true;
            }
        }
        return true;
    }
}
