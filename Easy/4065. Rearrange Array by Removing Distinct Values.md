You are given an integer array `nums`.

You start with an **empty** array `ans`. Repeat the following operation until `nums` is **empty**:

-   Identify **all** **distinct** values currently present in `nums`.
-   Remove **one** occurrence of every **distinct** value currently in `nums`, and append those values to `ans` in **ascending** order.

Return the array `ans`.

**Example 1:**
```
Input: nums = [3,1,3,2,1,3]
Output: [1,2,3,1,3,3]
Explanation:
```

**Example 2:**
```
Input: nums = [7,7,4,4,4]
Output: [4,7,4,7,4]
Explanation:
```

**Constraints:**

-   `1 <= nums.length <= 100`
-   `1 <= nums[i] <= 100`


### Python
```py
class Solution:
    def rearrangeArray(self, nums: list[int]) -> list[int]:
        counter = Counter(nums)
        result = []
        for _ in range(sum(counter.values())):
            _ = []
            for k in sorted(counter):
                result.append(k)
                counter[k] -= 1
                if counter[k] == 0:
                    _.append(k)
            for k in _:
                del counter[k]
        return result
                
```

```go
func rearrangeArray(nums []int) []int {
	counter := make(map[int]int)
	result := []int{}
	for _, num := range nums {
		counter[num]++
	}
	for len(counter) > 0 {
		keys := []int{}
		for k := range counter {
			keys = append(keys, k)
		}
		sort.Ints(keys)
		remove := []int{}
		for _, k := range keys {
			result = append(result, k)
			counter[k]--
			if counter[k] == 0 {
				remove = append(remove, k)
			}
		}
		for _, k := range remove {
			delete(counter, k)
		}
	}
	return result
}
```

### C#
```c#
public class Solution
{
    public int[] RearrangeArray(int[] nums)
    {
        Dictionary<int, int> counter = new ();
        List<int> result = new ();
        foreach (int num in nums)
        {
            if (counter.ContainsKey(num))
            {
                counter[num]++;
            }
            else
            {
                counter[num] = 1;
            }
        }
        while (counter.Count > 0)
        {
            List<int> keys = counter.Keys.ToList();
            keys.Sort();
            List<int> remove = new ();
            foreach (int k in keys)
            {
                result.Add(k);
                counter[k]--;
                if (counter[k] == 0)
                {
                    remove.Add(k);
                }
            }
            foreach (int k in remove)
            {
                counter.Remove(k);
            }
        }
        return result.ToArray();
    }
}
```
