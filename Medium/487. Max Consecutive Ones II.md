class Solution:
    def findMaxConsecutiveOnes(self, nums: list[int]) -> int:
        arr = [[nums[0], 1]]
        for i in range(1, len(nums)):
            if nums[i] == 1 and arr[-1][0] == 1:
                arr[-1][1] += 1
            else:
                arr.append([nums[i], 1])
        n = len(arr)
        result = arr[0][1] if arr[0][0] == 1 else 0
        for i, (num, count) in enumerate(arr):
            if num == 0:
                if i > 0 and arr[i - 1][0] == 1:
                    count += arr[i - 1][1]
                if i + 1 < n and arr[i + 1][0] == 1:
                    count += arr[i + 1][1]
                result = max(result, count)
        return result
