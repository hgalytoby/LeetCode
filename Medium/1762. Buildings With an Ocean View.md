class Solution:
    def findBuildings(self, heights: List[int]) -> List[int]:
        max_h, result = 0, []
        for i in range(len(heights) - 1, -1, -1):
            if heights[i] > max_h:
                max_h = heights[i]
                result.append(i)
        return result[::-1]
