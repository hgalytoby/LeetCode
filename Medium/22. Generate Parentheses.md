Given `n` pairs of parentheses, write a function to _generate all combinations of well-formed parentheses_.

**Example 1:**
```
Input: n = 3
Output: ["((()))","(()())","(())()","()(())","()()()"]
```

**Example 2:**
```
Input: n = 1
Output: ["()"]
```

**Constraints:**

-   `1 <= n <= 8`


### Python
```py
class Solution:
    def generateParenthesis(self, n: int) -> list[str]:
        def backtracking(left_depth, right_depth):
            if left_depth == n and right_depth == n:
                result.append(''.join(cur))
                return
        
            if left_depth < n:
                cur.append('(')
                backtracking(left_depth + 1, right_depth)
                cur.pop()

            if right_depth < left_depth:
                cur.append(')')
                backtracking(left_depth, right_depth + 1)
                cur.pop()


        result, cur = [], []
        backtracking(0, 0)
        return result
```

### Go
```go
func generateParenthesis(n int) []string {
	result := []string{}
	cur := []byte{}

	var backtracking func(int, int)

	backtracking = func(leftDepth int, rightDepth int) {
		if leftDepth == n && rightDepth == n {
			result = append(result, string(cur))
			return
		}

		if leftDepth < n {
			cur = append(cur, '(')
			backtracking(leftDepth+1, rightDepth)
			cur = cur[:len(cur)-1]
		}

		if rightDepth < leftDepth {
			cur = append(cur, ')')
			backtracking(leftDepth, rightDepth+1)
			cur = cur[:len(cur)-1]
		}
	}

	backtracking(0, 0)

	return result
}
```

public class Solution
{
    public IList<string> GenerateParenthesis(int n)
    {
        List<string> result = new ();
        List<char> cur = new ();

        void Backtracking(int leftDepth, int rightDepth)
        {
            if (leftDepth == n && rightDepth == n)
            {
                result.Add(new string(cur.ToArray()));
                return;
            }
            if (leftDepth < n)
            {
                cur.Add('(');
                Backtracking(leftDepth + 1, rightDepth);
                cur.RemoveAt(cur.Count - 1);
            }
            if (rightDepth < leftDepth)
            {
                cur.Add(')');
                Backtracking(leftDepth, rightDepth + 1);
                cur.RemoveAt(cur.Count - 1);
            }
        }

        Backtracking(0, 0);
        return result;
    }
}
