Given a string  `s`  containing only three types of characters:  `'('`,  `')'`  and  `'*'`, return  `true`  _if_  `s`  _is  **valid**_.

The following rules define a  **valid**  string:

-   Any left parenthesis  `'('`  must have a corresponding right parenthesis  `')'`.
-   Any right parenthesis  `')'`  must have a corresponding left parenthesis  `'('`.
-   Left parenthesis  `'('`  must go before the corresponding right parenthesis  `')'`.
-   `'*'`  could be treated as a single right parenthesis  `')'`  or a single left parenthesis  `'('`  or an empty string  `""`.

**Example 1:**
```
Input: s = "()"
Output: true
```

**Example 2:**
```
Input: s = "(*)"
Output: true
```

**Example 3:**
```
Input: s = "(*))"
Output: true
```

**Constraints:**

-   `1 <= s.length <= 100`
-   `s[i]`  is  `'('`,  `')'`  or  `'*'`.


### Python
```python
class Solution:
    def checkValidString(self, s: str) -> bool:
        stack, star = [], []
        for i, c in enumerate(s):
            if c == ')':
                if not stack:
                    if not star:
                        return False
                    star.pop()
                else:
                    stack.pop()
            elif c == '(':
                stack.append(i)
            else:
                star.append(i)
        while stack and star:
            if stack.pop() > star.pop():
                return False
        return not bool(stack)
```

### Go
```go
func checkValidString(s string) bool {
    stack := []int{}
	star := []int{}
	for i, c := range s {
		if c == ')' {
			if len(stack) == 0 {
				if len(star) == 0 {
					return false
				}

				star = star[:len(star)-1]
			} else {
				stack = stack[:len(stack)-1]
			}
		} else if c == '(' {
			stack = append(stack, i)
		} else {
			star = append(star, i)
		}
	}
	for len(stack) > 0 && len(star) > 0 {
		stackIndex := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		starIndex := star[len(star)-1]
		star = star[:len(star)-1]
		if stackIndex > starIndex {
			return false
		}
	}
	return len(stack) == 0
}
```

```c#
public class Solution {
    public bool CheckValidString(string s) {
        List<int> stack = new ();
        List<int> star = new ();
        for (int i = 0; i < s.Length; i++)
        {
            char c = s[i];
            if (c == ')')
            {
                if (stack.Count == 0)
                {
                    if (star.Count == 0)
                    {
                        return false;
                    }
                    star.RemoveAt(star.Count - 1);
                }
                else
                {
                    stack.RemoveAt(stack.Count - 1);
                }
            }
            else if (c == '(')
            {
                stack.Add(i);
            }
            else
            {
                star.Add(i);
            }
        }
        while (stack.Count > 0 && star.Count > 0)
        {
            int stackIndex = stack[stack.Count - 1];
            stack.RemoveAt(stack.Count - 1);
            int starIndex = star[star.Count - 1];
            star.RemoveAt(star.Count - 1);
            if (stackIndex > starIndex)
            {
                return false;
            }
        }
        return stack.Count == 0;
    }
}
```
