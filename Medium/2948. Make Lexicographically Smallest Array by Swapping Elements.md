You are given a  **0-indexed**  array of  **positive**  integers  `nums`  and a  **positive**  integer  `limit`.

In one operation, you can choose any two indices  `i`  and  `j`  and swap  `nums[i]`  and  `nums[j]`  **if**  `|nums[i] - nums[j]| <= limit`.

Return  _the  **lexicographically smallest array**  that can be obtained by performing the operation any number of times_.

An array  `a`  is lexicographically smaller than an array  `b`  if in the first position where  `a`  and  `b`  differ, array  `a`  has an element that is less than the corresponding element in  `b`. For example, the array  `[2,10,3]`  is lexicographically smaller than the array  `[10,2,3]`  because they differ at index  `0`  and  `2 < 10`.

**Example 1:**
```
Input: nums = [1,5,3,9,8], limit = 2
Output: [1,3,5,8,9]
Explanation: Apply the operation 2 times:
- Swap nums[1] with nums[2]. The array becomes [1,3,5,9,8]
- Swap nums[3] with nums[4]. The array becomes [1,3,5,8,9]
We cannot obtain a lexicographically smaller array by applying any more operations.
Note that it may be possible to get the same result by doing different operations.
```

**Example 2:**
```
Input: nums = [1,7,6,18,2,1], limit = 3
Output: [1,6,7,18,1,2]
Explanation: Apply the operation 3 times:
- Swap nums[1] with nums[2]. The array becomes [1,6,7,18,2,1]
- Swap nums[0] with nums[4]. The array becomes [2,6,7,18,1,1]
- Swap nums[0] with nums[5]. The array becomes [1,6,7,18,1,2]
We cannot obtain a lexicographically smaller array by applying any more operations.
```

**Example 3:**
```
Input: nums = [1,7,28,19,10], limit = 3
Output: [1,7,28,19,10]
Explanation: [1,7,28,19,10] is the lexicographically smallest array we can obtain because we cannot apply the operation on any two indices.
```

**Constraints:**

-   `1 <= nums.length <= 105`
-   `1 <= nums[i] <= 109`
-   `1 <= limit <= 109`


### Python
```py
class Solution:
    def lexicographicallySmallestArray(self, nums: List[int], limit: int) -> List[int]:
        n = len(nums)
        arr = [[nums[i], i] for i in range(n)]
        arr.sort()
        group = [[arr[0]]]
        result = [0] * n
        for i in range(1, n):
            if arr[i][0] - limit <= group[-1][-1][0]:
                group[-1].append(arr[i])
            else:
                group.append([arr[i]])
        index_group = [sorted(index for _, index in v) for v in group]
        for i in range(len(group)):
            for j in range(len(group[i])):
                num, index = group[i][j][0], index_group[i][j]
                result[index] = num
        return result

```


### Go
```go
func lexicographicallySmallestArray(nums []int, limit int) []int {
    n := len(nums)
	result := make([]int, n)
	arr := make([][2]int, n)
	for i := 0; i < n; i++ {
		arr[i] = [2]int{nums[i], i}
	}
	sort.Slice(arr, func(i, j int) bool {
		if arr[i][0] == arr[j][0] {
			return arr[i][1] < arr[j][1]
		}
		return arr[i][0] < arr[j][0]
	})
	group := [][][2]int{{arr[0]}}
	for i := 1; i < n; i++ {
		lastGroup := group[len(group)-1]
		lastElement := lastGroup[len(lastGroup)-1]
		
		if arr[i][0]-limit <= lastElement[0] {
			group[len(group)-1] = append(group[len(group)-1], arr[i])
		} else {
			group = append(group, [][2]int{arr[i]})
		}
	}
	for i := 0; i < len(group); i++ {
		currentGroup := group[i]
		m := len(currentGroup)
		indexGroup := make([]int, m)
		for j := 0; j < m; j++ {
			indexGroup[j] = currentGroup[j][1]
		}
		sort.Ints(indexGroup)
		for j := 0; j < m; j++ {
			num := currentGroup[j][0]
			index := indexGroup[j]
			result[index] = num
		}
    }
	return result
}
```

### C#
```c#
public class Solution {
    public int[] LexicographicallySmallestArray(int[] nums, int limit) {
        int n = nums.Length;
        List<(int Num, int Idx)> arr = new (n);
        int[] result = new int[n];
        for (int i = 0; i < n; i++) {
            arr.Add((nums[i], i));
        }
        arr.Sort((a, b) => a.Num == b.Num ? a.Idx.CompareTo(b.Idx) : a.Num.CompareTo(b.Num));
        List<List<(int Num, int Idx)>> group = new () {
            new List<(int Num, int Idx)> { arr[0] }
        };
        for (int i = 1; i < n; i++) {
            List<(int Num, int Idx)> lastGroup = group[group.Count - 1];
            (int Num, int Idx) lastElement = lastGroup[lastGroup.Count - 1];
            if (arr[i].Num - limit <= lastElement.Num) {
                lastGroup.Add(arr[i]);
            } else {
                group.Add(new List<(int Num, int Idx)> { arr[i] });
            }
        }
        for (int i = 0; i < group.Count; i++) {
            List<(int Num, int Idx)> currentGroup = group[i];
            List<int> indexGroup = currentGroup.Select(x => x.Idx).OrderBy(idx => idx).ToList();
            for (int j = 0; j < currentGroup.Count; j++) {
                int num = currentGroup[j].Num;
                int index = indexGroup[j];
                result[index] = num;
            }
        }
        return result;
    }
}
```
