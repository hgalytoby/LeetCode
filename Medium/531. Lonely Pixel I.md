```py
class Solution:
    def findLonelyPixel(self, picture: List[List[str]]) -> int:
        rows, cols = Counter(), Counter()
        m, n = len(picture), len(picture[0])
        b = []
        for row in range(m):
            for col in range(n):
                if picture[row][col] == 'B':
                    rows[row] += 1
                    cols[col] += 1
                    b.append((row, col))
        return sum(rows[row] == 1 and cols[col] == 1 for row, col in b)
```

```go
func findLonelyPixel(picture [][]byte) int {
    m, n := len(picture), len(picture[0])
	rows := make([]int, m)
	cols := make([]int, n)
	b := [][2]int{}
	result := 0
	for row := 0; row < m; row++ {
		for col := 0; col < n; col++ {
			if picture[row][col] == 'B' {
				rows[row]++
				cols[col]++
				b = append(b, [2]int{row, col})
			}
		}
	}
	for _, pos := range b {
		row, col := pos[0], pos[1]
		if rows[row] == 1 && cols[col] == 1 {
			result++
		}
	}
	return result
}
```

```c#
    public int FindLonelyPixel(char[][] picture) {
        int m = picture.Length;

        List<(int row, int col)> b = new ();

                {

                    cols[col]++;
                    b.Add((row, col));
                }
            }
        }
        foreach ((int row, int col) in b)
        {
            if (rows[row] == 1 && cols[col] == 1)
            {

            }
        }

    }
```
