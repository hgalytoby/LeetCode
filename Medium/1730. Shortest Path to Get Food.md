```py
class Solution:
    D = [(0, 1), (0, -1), (1, 0), (-1, 0)]

    def getFood(self, grid: List[List[str]]) -> int:
        m, n = len(grid), len(grid[0])
        queue = deque([])
        for r in range(m):
            for c in range(n):
                if grid[r][c] == '*':
                    queue.append((r, c, 0))
                    break
        visited = {(queue[0][0], queue[0][1])}
        while queue:
            x, y, v = queue.popleft()
            next_v = v + 1
            for dx, dy in self.D:
                next_x, next_y = x + dx, y + dy
                if next_x >= m or next_x < 0 or next_y >= n or next_y < 0 or grid[next_x][next_y] == 'X' or (next_x, next_y) in visited:
                    continue
                if grid[next_x][next_y] == '#':
                    return next_v
                visited.add((next_x, next_y))
                queue.append((next_x, next_y, next_v))
        return -1
```

```go
func getFood(grid [][]byte) int {
    type Node struct {
		x, y, dist int
	}

    m, n := len(grid), len(grid[0])
	directions := [][2]int{
		{0, 1},
		{0, -1},
		{1, 0},
		{-1, 0},
	}
	queue := []Node{}
	for r := 0; r < m; r++ {
		for c := 0; c < n; c++ {
			if grid[r][c] == '*' {
				queue = append(queue, Node{r, c, 0})
				break
			}
		}
	}
	visited := make([][]bool, m)
	for i := range visited {
		visited[i] = make([]bool, n)
	}
	visited[queue[0].x][queue[0].y] = true
	for head := 0; head < len(queue); head++ {
		node := queue[head]
		nextV := node.dist + 1
		for _, d := range directions {
			nextX := node.x + d[0]
			nextY := node.y + d[1]
			if nextX >= m || nextX < 0 ||
				nextY >= n || nextY < 0 ||
				grid[nextX][nextY] == 'X' ||
				visited[nextX][nextY] {
				continue
			}
			if grid[nextX][nextY] == '#' {
				return nextV
			}
			visited[nextX][nextY] = true
			queue = append(queue, Node{nextX, nextY, nextV})
		}
	}
	return -1
}
```

### Go
```go
public class Solution {
    private static readonly int[][] D = new int[][] {
        new int[] { 0, 1 },
        new int[] { 0, -1 },
        new int[] { 1, 0 },
        new int[] { -1, 0 }
    };

    public int GetFood(char[][] grid) {
        int m = grid.Length;
        int n = grid[0].Length;
        Queue<(int x, int y, int dist)> queue = new ();
        for (int r = 0; r < m; r++)
        {
            for (int c = 0; c < n; c++)
            {
                if (grid[r][c] == '*')
                {
                    queue.Enqueue((r, c, 0));
                    break;
                }
            }
        }
        bool[,] visited = new bool[m, n];
        (int x, int y, int dist) start = queue.Peek();
        visited[start.x, start.y] = true;
        while (queue.Count > 0)
        {
            (int x, int y, int dist) node = queue.Dequeue();
            int nextV = node.dist + 1;
            foreach (int[] d in D)
            {
                int nextX = node.x + d[0];
                int nextY = node.y + d[1];
                if (nextX >= m || nextX < 0 ||
                    nextY >= n || nextY < 0 ||
                    grid[nextX][nextY] == 'X' ||
                    visited[nextX, nextY])
                {
                    continue;
                }
                if (grid[nextX][nextY] == '#')
                {
                    return nextV;
                }
                visited[nextX, nextY] = true;
                queue.Enqueue((nextX, nextY, nextV));
            }
        }
        return -1;
    }
}
```
